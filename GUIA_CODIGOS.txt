class FacebookPost(models.Model):
    """
    Almacena publicaciones realizadas desde la aplicación
    """
    # Relación N:1 con FacebookAccount
    account = models.ForeignKey(
        FacebookAccount, 
        on_delete=models.CASCADE,
        related_name='publicaciones',
        help_text="Cuenta que realizó la publicación"
    )
    
    # Contenido
    mensaje = models.TextField(
        help_text="Texto del post"
    )
    
    # ID del post en Facebook
    post_id = models.CharField(
        max_length=200, 
        blank=True,
        help_text="ID asignado por Facebook"
    )
    
    # Fechas
    fecha_publicacion = models.DateTimeField(
        auto_now_add=True,
        help_text="Cuándo se publicó"
    )
    
    fecha_actualizacion = models.DateTimeField(
        auto_now=True,
        help_text="Última actualización de métricas"
    )
    
    # Métricas de interacción
    likes = models.IntegerField(
        default=0,
        help_text="Cantidad de Me Gusta"
    )
    
    comentarios = models.IntegerField(
        default=0,
        help_text="Cantidad de comentarios"
    )
    
    compartidos = models.IntegerField(
        default=0,
        help_text="Cantidad de compartidos"
    )
    
    alcance = models.IntegerField(
        default=0,
        help_text="Personas únicas que vieron el post"
    )
    
    # Estado del post
    ESTADO_CHOICES = [
        ('borrador', 'Borrador'),
        ('publicado', 'Publicado'),
        ('eliminado', 'Eliminado'),
        ('error', 'Error al publicar'),
    ]
    
    estado = models.CharField(
        max_length=20,
        choices=ESTADO_CHOICES,
        default='publicado'
    )
    
    class Meta:
        verbose_name = 'Publicación de Facebook'
        verbose_name_plural = 'Publicaciones de Facebook'
        ordering = ['-fecha_publicacion']
        indexes = [
            models.Index(fields=['-fecha_publicacion']),
            models.Index(fields=['post_id']),
        ]
    
    def __str__(self):
        return f"Post: {self.mensaje[:50]}..." if len(self.mensaje) > 50 else f"Post: {self.mensaje}"
    
    @property
    def engagement_total(self):
        """Calcula engagement total"""
        return self.likes + self.comentarios + self.compartidos
    
    @property
    def tasa_engagement(self):
        """Porcentaje de engagement respecto al alcance"""
        if self.alcance > 0:
            return round((self.engagement_total / self.alcance) * 100, 2)
        return 0.0

def facebook_login(request):
    """
    Paso 1 del flujo OAuth 2.0:
    Redirige al usuario a Facebook para autorizar nuestra app
    """
    
    # Construir URL de autorización
    auth_url = "https://www.facebook.com/v18.0/dialog/oauth?"
    auth_url += f"client_id={settings.FACEBOOK_APP_ID}"
    auth_url += f"&redirect_uri={settings.FACEBOOK_REDIRECT_URI}"
    auth_url += "&scope=email,public_profile,pages_show_list,pages_read_engagement,pages_manage_posts"
    auth_url += "&response_type=code"
    
    # PARÁMETROS:
    # client_id: App ID de Facebook Developers
    # redirect_uri: URL donde Facebook redirige (callback)
    # scope: Permisos solicitados
    #   - email: acceso al correo
    #   - public_profile: nombre, foto, etc
    #   - pages_show_list: listar páginas
    #   - pages_read_engagement: leer métricas
    #   - pages_manage_posts: PUBLICAR en páginas ⭐
    # response_type: "code" para OAuth flow
    
    return redirect(auth_url)

def facebook_callback(request):
    """
    Paso 2 y 3 del flujo OAuth:
    - Paso 2: Recibir código de autorización
    - Paso 3: Intercambiar código por access_token
    """
    
    # PASO 2.1: Obtener código de la URL
    code = request.GET.get('code')
    
    if not code:
        error_description = request.GET.get('error_description', 'Usuario canceló')
        return render(request, 'error.html', {
            'error': 'No se recibió código',
            'detalle': error_description
        })
    
    # PASO 2.2: Intercambiar código por access_token
    token_url = "https://graph.facebook.com/v18.0/oauth/access_token"
    
    params = {
        'client_id': settings.FACEBOOK_APP_ID,
        'client_secret': settings.FACEBOOK_APP_SECRET,
        'redirect_uri': settings.FACEBOOK_REDIRECT_URI,
        'code': code,
    }
    
    response = requests.get(token_url, params=params)
    token_data = response.json()
    
    if 'error' in token_data:
        return render(request, 'error.html', {
            'error': 'Error al obtener token',
            'detalle': token_data.get('error_description')
        })
    
    access_token = token_data.get('access_token')
    
    # PASO 2.3: Obtener información del usuario
    me_url = "https://graph.facebook.com/v18.0/me"
    
    user_params = {
        'fields': 'id,name,email,picture.type(large)',
        'access_token': access_token,
    }
    
    user_response = requests.get(me_url, params=user_params)
    user_data = user_response.json()
    
    if 'error' in user_data:
        return render(request, 'error.html', {
            'error': 'Error al obtener datos del usuario',
            'detalle': user_data.get('error', {}).get('message')
        })
    
    # PASO 2.4: Crear o actualizar cuenta
    try:
        facebook_account = FacebookAccount.objects.get(facebook_id=user_data['id'])
        # Actualizar token y datos
        facebook_account.access_token = access_token
        facebook_account.nombre = user_data['name']
        facebook_account.email = user_data.get('email', '')
        facebook_account.foto_perfil = user_data['picture']['data']['url']
        facebook_account.save()
        
    except FacebookAccount.DoesNotExist:
        # Crear nuevo usuario Django
        if request.user.is_authenticated:
            django_user = request.user
        else:
            username = f"fb_{user_data['id']}"
            django_user, created = User.objects.get_or_create(
                username=username,
                defaults={
                    'email': user_data.get('email', ''),
                    'first_name': user_data['name'].split()[0] if user_data['name'] else '',
                }
            )
        
        # Crear cuenta Facebook
        facebook_account = FacebookAccount.objects.create(
            user=django_user,
            facebook_id=user_data['id'],
            access_token=access_token,
            nombre=user_data['name'],
            email=user_data.get('email', ''),
            foto_perfil=user_data['picture']['data']['url'],
        )
    
    # PASO 2.5: Iniciar sesión en Django
    login(request, facebook_account.user, backend='django.contrib.auth.backends.ModelBackend')
    
    return redirect('dashboard')

@login_required
def dashboard(request):
    """
    Muestra panel con información del usuario y sus publicaciones
    """
    try:
        facebook_account = FacebookAccount.objects.get(user=request.user)
        
        # Obtener últimas 10 publicaciones
        posts = FacebookPost.objects.filter(account=facebook_account)[:10]
        
        # Calcular estadísticas
        total_posts = FacebookPost.objects.filter(account=facebook_account).count()
        total_likes = sum(post.likes for post in posts)
        total_comentarios = sum(post.comentarios for post in posts)
        
        context = {
            'facebook_account': facebook_account,
            'posts': posts,
            'total_posts': total_posts,
            'total_likes': total_likes,
            'total_comentarios': total_comentarios,
        }
        
        return render(request, 'dashboard.html', context)
        
    except FacebookAccount.DoesNotExist:
        return redirect('facebook_login')

@login_required
def publicar_facebook(request):
    """
    Publica en una PÁGINA de Facebook
    (No es posible publicar en perfil personal desde 2018)
    """
    
    if request.method == 'POST':
        mensaje = request.POST.get('mensaje', '').strip()
        
        if not mensaje:
            return render(request, 'publicar.html', {
                'error': 'El mensaje no puede estar vacío'
            })
        
        try:
            account = FacebookAccount.objects.get(user=request.user)
            
            # PASO 1: Obtener lista de PÁGINAS administradas
            pages_url = "https://graph.facebook.com/v18.0/me/accounts"
            pages_params = {'access_token': account.access_token}
            
            pages_response = requests.get(pages_url, params=pages_params)
            pages_data = pages_response.json()
            
            if 'error' in pages_data:
                return render(request, 'publicar.html', {
                    'error': f"Error: {pages_data['error']['message']}",
                    'detalle': 'Necesitas autorizar "pages_manage_posts"',
                    'solucion': 'Cierra sesión y vuelve a iniciar sesión'
                })
            
            if not pages_data.get('data'):
                return render(request, 'publicar.html', {
                    'error': 'No administras ninguna Página',
                    'detalle': 'Crea una Página de prueba',
                    'solucion': 'Visita https://facebook.com/pages/create'
                })
            
            # PASO 2: Usar primera página
            primera_pagina = pages_data['data'][0]
            page_id = primera_pagina['id']
            page_access_token = primera_pagina['access_token']
            page_name = primera_pagina['name']
            
            # PASO 3: Publicar en la PÁGINA
            post_url = f"https://graph.facebook.com/v18.0/{page_id}/feed"
            
            data = {
                'message': mensaje,
                'access_token': page_access_token,  # ⭐ Token de la PÁGINA
            }
            
            response = requests.post(post_url, data=data)
            post_data = response.json()
            
            if 'error' in post_data:
                error_message = post_data['error']['message']
                error_code = post_data['error'].get('code')
                
                return render(request, 'publicar.html', {
                    'error': f"Error {error_code}: {error_message}",
                    'detalle': 'Revisa los permisos',
                    'json_response': json.dumps(post_data, indent=2)
                })
            
            # PASO 4: Guardar en base de datos
            FacebookPost.objects.create(
                account=account,
                mensaje=mensaje,
                post_id=post_data.get('id', ''),
                estado='publicado',
            )
            
            post_facebook_url = f"https://www.facebook.com/{post_data.get('id', '').replace('_', '/posts/')}"
            
            # PASO 5: Éxito
            return render(request, 'publicar.html', {
                'success': f'✅ Publicación exitosa en "{page_name}"',
                'post_id': post_data.get('id', ''),
                'page_name': page_name,
                'post_url': post_facebook_url,
                'nota': 'Se publicó en tu PÁGINA, no en tu perfil'
            })
        
        except FacebookAccount.DoesNotExist:
            return redirect('facebook_login')
        except requests.exceptions.RequestException as e:
            return render(request, 'publicar.html', {
                'error': 'Error de conexión',
                'detalle': str(e)
            })
        except Exception as e:
            return render(request, 'publicar.html', {
                'error': f'Error: {type(e).__name__}',
                'detalle': str(e)
            })
    
    return render(request, 'publicar.html')

@login_required
def actualizar_metricas_post(request, post_id):
    """
    Actualiza métricas (likes, comentarios, shares) de una publicación
    consultando Facebook Graph API
    """
    
    try:
        # Obtener el post
        post = FacebookPost.objects.get(id=post_id, account__user=request.user)
        
        # Consultar métricas en Facebook
        insights_url = f"https://graph.facebook.com/v18.0/{post.post_id}"
        params = {
            'fields': 'reactions.summary(true),comments.summary(true),shares',
            'access_token': post.account.access_token,
        }
        
        response = requests.get(insights_url, params=params)
        metrics_data = response.json()
        
        # Actualizar en base de datos
        post.likes = metrics_data.get('reactions', {}).get('summary', {}).get('total_count', 0)
        post.comentarios = metrics_data.get('comments', {}).get('summary', {}).get('total_count', 0)
        post.compartidos = metrics_data.get('shares', {}).get('count', 0)
        post.save()
        
        return JsonResponse({
            'success': True,
            'likes': post.likes,
            'comentarios': post.comentarios,
            'compartidos': post.compartidos,
        })
        
    except FacebookPost.DoesNotExist:
        return JsonResponse({'error': 'Post no encontrado'}, status=404)
    except Exception as e:
        return JsonResponse({'error': str(e)}, status=500)

FLUJO DE TRABAJO COMPLETO
================================================================================

1️⃣  USUARIO VISITA LA APP
   └─→ urls.py (inicio) → views.inicio() → login.html


2️⃣  USUARIO HACE CLIC EN "LOGIN CON FACEBOOK"
   └─→ urls.py (facebook/login/) → views.facebook_login() 
       └─→ REDIRIGE A: https://facebook.com/dialog/oauth?...


3️⃣  USUARIO AUTORIZA LA APP EN FACEBOOK
   └─→ Facebook redirige a: /facebook/callback/?code=ABC123


4️⃣  PROCESAR CALLBACK
   └─→ urls.py (facebook/callback/) → views.facebook_callback()
       ├─→ Obtiene code de URL
       ├─→ Intercambia code por access_token
       ├─→ Obtiene datos del usuario
       ├─→ Crea FacebookAccount en BD
       ├─→ Inicia sesión en Django
       └─→ REDIRIGE A: /dashboard/


5️⃣  MOSTRAR DASHBOARD
   └─→ urls.py (dashboard/) → views.dashboard() → dashboard.html
       ├─→ Obtiene FacebookAccount del usuario
       ├─→ Obtiene últimos posts
       ├─→ Calcula estadísticas
       └─→ Muestra información


6️⃣  USUARIO CREA PUBLICACIÓN
   └─→ urls.py (facebook/publicar/) → views.publicar_facebook()
       ├─→ GET: Muestra formulario (publicar.html)
       ├─→ POST: Procesa publicación
       │   ├─→ Obtiene lista de PÁGINAS del usuario
       │   ├─→ Publica en la PÁGINA (no perfil)
       │   ├─→ Guarda en BD
       │   └─→ Muestra éxito
       └─→ Vuelve a mostrar formulario


7️⃣  ACTUALIZAR MÉTRICAS (OPCIONAL)
   └─→ API: /facebook/post/<id>/metricas/ → views.actualizar_metricas_post()
       ├─→ Consulta Facebook Graph API
       ├─→ Obtiene likes, comentarios, shares
       ├─→ Actualiza BD
       └─→ Retorna JSON con métricas

FLUJO OAUTH 2.0 CON FACEBOOK (TÉCNICO)

PASO 1: AUTHORIZATION REQUEST (facebook_login)
┌─────────────────────────────────────────────────────────────┐
│ Usuario hace clic en "Login con Facebook"                   │
└─────────────────────────────────────────────────────────────┘
        ↓
URL: https://www.facebook.com/v18.0/dialog/oauth?
        client_id=2067776030644149
        &redirect_uri=http://localhost:8000/facebook/callback/
        &scope=email,public_profile,pages_show_list,pages_read_engagement,pages_manage_posts
        &response_type=code


PASO 2: USER AUTHORIZATION
┌─────────────────────────────────────────────────────────────┐
│ Facebook muestra pantalla de autorización                   │
│ Usuario ve qué permisos solicita la app                     │
│ Usuario hace clic en "Continuar como [Nombre]"             │
└─────────────────────────────────────────────────────────────┘
        ↓
Facebook redirige a:
        http://localhost:8000/facebook/callback/?code=CODIGO123&state=...


PASO 3: AUTHORIZATION CODE (facebook_callback - parte 1)
┌─────────────────────────────────────────────────────────────┐
│ views.facebook_callback() recibe el code                    │
│ code = request.GET.get('code')                              │
└─────────────────────────────────────────────────────────────┘
        ↓
code = CODIGO123


PASO 4: ACCESS TOKEN REQUEST (facebook_callback - parte 2)
┌─────────────────────────────────────────────────────────────┐
│ SERVIDOR envía petición POST a Facebook                     │
│ (NO es visible para el usuario)                             │
└─────────────────────────────────────────────────────────────┘
        ↓
POST https://graph.facebook.com/v18.0/oauth/access_token?
        client_id=APP_ID
        &client_secret=APP_SECRET
        &redirect_uri=http://localhost:8000/facebook/callback/
        &code=CODIGO123

Respuesta:
{
    "access_token": "EAACG...",
    "token_type": "bearer",
    "expires_in": 5183999
}


PASO 5: GET USER DATA (facebook_callback - parte 3)
┌─────────────────────────────────────────────────────────────┐
│ SERVIDOR obtiene datos del usuario con el token             │
└─────────────────────────────────────────────────────────────┘
        ↓
GET https://graph.facebook.com/v18.0/me?
        fields=id,name,email,picture.type(large)
        &access_token=EAACG...

Respuesta:
{
    "id": "1234567890",
    "name": "Juan Pérez",

    "picture": {
        "data": {
            "url": "https://..."
        }
    }
}

│ SERVIDOR crea usuario Django y FacebookAccount              │
│ SERVIDOR inicia sesión del usuario                          │
│ SERVIDOR redirige al dashboard                              │
└─────────────────────────────────────────────────────────────┘

================================================================================
ERRORES COMUNES Y SOLUCIONES
================================================================================

❌ Error: "Redirect URI mismatch"
✅ Solución: Asegúrate que en Facebook Developers esté registrada exactamente:
   http://localhost:8000/facebook/callback/
   (Sin / al final si lo tienes con /)

❌ Error: "Invalid OAuth access token"
✅ Solución: El token ha expirado. Pedir al usuario que vuelva a iniciar sesión.

❌ Error: "User canceled Login"
✅ Solución: Normal. El usuario cerró la ventana. Mostrar error amigable.

❌ Error: "No Facebook pages found"
✅ Solución: El usuario no administra ninguna página. Pedir que cree una página de prueba.

❌ Error: "DatabaseError: (1054, "Unknown column 'usuario_id'")"
✅ Solución: Ejecutar migraciones:
   python manage.py migrate

❌ Error: "TemplateDoesNotExist"
✅ Solución: Asegúrate que las carpetas estén en:
   social_app/templates/base.html
   social_app/templates/login.html
   etc.

    list_display = ('mensaje_corto', 'account', 'fecha_publicacion', 'estado', 'likes')
    list_filter = ('estado', 'fecha_publicacion')
    readonly_fields = ('fecha_publicacion', 'fecha_actualizacion')
    
    def mensaje_corto(self, obj):
        return obj.mensaje[:50] + "..." if len(obj.mensaje) > 50 else obj.mensaje
